    // Create subtle floating particles

    // Show toast notification
    function showToast(message, duration = 3000) {
      const toast = document.getElementById('toast');
      const toastMessage = document.getElementById('toast-message');

      toastMessage.textContent = message;
      toast.classList.add('show');

      setTimeout(() => {
        toast.classList.remove('show');
      }, duration);
    }

    // Handle copy functionality with improved feedback
    function copyToClipboard(text, button) {
      navigator.clipboard.writeText(text).then(() => {
        // Update button state
        button.classList.add('copied');

        // Change button text and icon
        button.innerHTML = `
          <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
            <path d="M20 6L9 17l-5-5"></path>
          </svg>
          <span>Copied!</span>`;

        // Show toast notification
        showToast('Text copied to clipboard!');

        // Reset button after delay
        setTimeout(() => {
          button.classList.remove('copied');
          button.innerHTML = `
            <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
              <rect x="9" y="9" width="13" height="13" rx="2" ry="2"></rect>
              <path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1"></path>
            </svg>
            <span>Copy</span>`;
        }, 2000);
      }).catch(err => {
        console.error('Could not copy text: ', err);
        showToast('Failed to copy text', 3000);
      });
    }

    // Subtle service selection: add a border to the selected option.

      document.getElementById('response-section').classList.add('hidden');

    }

    function handleImageUpload(input) {
      if (input.files.length) {
        const file = input.files[0];
        // Validate file type and size
        if (!file.type.match('image.*')) {
          showToast('Please select an image file', 3000);
          clearImageUpload();
          return;
        }

        // Max file size: 5MB
        if (file.size > 5 * 1024 * 1024) {
          showToast('Image size should be less than 5MB', 3000);
          clearImageUpload();
          return;
        }

        const imageURL = URL.createObjectURL(file);
        const preview = document.getElementById('image-preview');
        const previewContainer = document.getElementById('image-preview-container');
        preview.src = imageURL;
        previewContainer.classList.remove('hidden');
        document.getElementById('analyze-button').disabled = false;
        storeRecentActivity('image', imageURL);
      }
    }

    function clearImageUpload() {
      document.getElementById('image-input').value = '';
      document.getElementById('image-preview-container').classList.add('hidden');
      document.getElementById('analyze-button').disabled = true;
    }

    // Enhanced process response function
    function processNewResponse(node) {
      if (!node || node.nodeType !== 1) return;

      let fullText = node.textContent.trim();
      // Cleanup text formatting if needed
      if (fullText.toLowerCase().startsWith("corrected text:")) {
        fullText = fullText.substring("corrected text:".length).trim();
      }

      // Create a wrapper to position the copy button
      const wrapper = document.createElement('div');
      wrapper.className = 'relative message-in response-wrapper';

      // Create a pre/code block for markdown styling
      const preBlock = document.createElement('pre');
      preBlock.className = 'response-block';
      preBlock.style.overflowX = 'auto';
      const codeBlock = document.createElement('code');
      codeBlock.className = 'language-md';
      codeBlock.innerText = fullText;
      preBlock.appendChild(codeBlock);
      wrapper.appendChild(preBlock);

      // Create the enhanced copy button
      const copyBtn = document.createElement('button');
      copyBtn.className = 'copy-btn';
      copyBtn.setAttribute('aria-label', 'Copy to clipboard');
      copyBtn.innerHTML = `
        <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
          <rect x="9" y="9" width="13" height="13" rx="2" ry="2"></rect>
          <path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1"></path>
        </svg>
        <span>Copy</span>`;

      // Improved click handler with enhanced feedback
      copyBtn.addEventListener('click', (e) => {
        e.preventDefault();
        e.stopPropagation();
        copyToClipboard(codeBlock.textContent, copyBtn);
      });

      // Add keyboard accessibility
      copyBtn.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          copyToClipboard(codeBlock.textContent, copyBtn);
        }
      });

      // Make the button focusable for keyboard navigation
      copyBtn.setAttribute('tabindex', '0');

      wrapper.appendChild(copyBtn);

      // Replace the original node with our enhanced version
      node.parentNode.replaceChild(wrapper, node);

      // Add copy on double-click for the entire response block
      preBlock.addEventListener('dblclick', (e) => {
        e.preventDefault();
        copyToClipboard(codeBlock.textContent, copyBtn);
      });
    }

    document.body.addEventListener('htmx:afterRequest', (event) => {
      const form = event.detail.elt;

      const submitBtn = form.querySelector('button[type="submit"]');

      if (spinner) spinner.classList.remove('active');
      if (submitBtn) submitBtn.disabled = false;

      // Handle request errors
      if (event.detail.successful) {
        document.getElementById('response-section').classList.remove('hidden');

        // Process the newly added response for styling and copy functionality
        const responseContainer = document.getElementById('response-container');
        processNewResponse(responseContainer.firstElementChild);
      } else {
        showToast('Request failed. Please try again.', 4000);
      }
    });

    // Store recent activity in local storage
    function storeRecentActivity(type, content) {
      try {
        let recentActivities = localStorage.getItem(`recent-${type}`) ?
          JSON.parse(localStorage.getItem(`recent-${type}`)) : [];

        // Create new activity with timestamp
        const newActivity = {
          id: Date.now(),
          content: content,
          timestamp: new Date().toISOString()
        };

        // Add to beginning of array and limit to 10 items
        recentActivities.unshift(newActivity);
        if (recentActivities.length > 10) {
          recentActivities = recentActivities.slice(0, 10);
        }

        localStorage.setItem(`recent-${type}`, JSON.stringify(recentActivities));

        // Update sidebar if it's the current view
        updateRecentSidebar(type);
      } catch (e) {
        console.error('Error storing recent activity:', e);
      }
    }

    // Load and display recent activities in sidebar
    function loadRecentActivities() {
      showRecent('text'); // Default to text tab
    }

    function showRecent(type) {
      // Update tab selection styling
      document.querySelectorAll('[id^="btn-recent-"]').forEach(btn => {
        btn.classList.remove('border-[#F0EDCC]', 'text-[#F0EDCC]');
        btn.classList.add('border-transparent', 'text-[rgba(240,237,204,0.7)]');
      });
      document.getElementById(`btn-recent-${type}s`).classList.remove('border-transparent', 'text-[rgba(240,237,204,0.7)]');
      document.getElementById(`btn-recent-${type}s`).classList.add('border-[#F0EDCC]', 'text-[#F0EDCC]');

      // Hide all sections and show the selected one
      document.querySelectorAll('.recent-section').forEach(section => {
        section.classList.add('hidden');
      });
      document.getElementById(`recent-${type}s`).classList.remove('hidden');

      updateRecentSidebar(type);
    }

    function updateRecentSidebar(type) {
      try {
        const container = document.getElementById(`recent-${type}s`);
        const recentItems = localStorage.getItem(`recent-${type}`) ?
          JSON.parse(localStorage.getItem(`recent-${type}`)) : [];

        if (recentItems.length === 0) {
          container.innerHTML = `<p class="text-center py-4 text-[rgba(240,237,204,0.7)]">No recent ${type} activities</p>`;
          return;
        }

        let html = '<ul class="recent-list">';

        recentItems.forEach(item => {
          const date = new Date(item.timestamp);
          const formattedDate = `${date.toLocaleDateString()} ${date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}`;

          if (type === 'text') {
            // For text, show preview
            const preview = item.content.length > 30 ?
              item.content.substring(0, 30) + '...' : item.content;

            html += `
          <li onclick="loadPreviousActivity('text', '${item.id}')" class="rounded hover:bg-[rgba(240,237,204,0.1)] p-2">
            <div class="flex flex-col">
              <span class="text-sm font-medium">${preview}</span>
              <span class="text-xs text-[rgba(240,237,204,0.7)]">${formattedDate}</span>
            </div>
            <button onclick="event.stopPropagation(); deleteActivity('text', ${item.id})" class="p-1 rounded-full hover:bg-[rgba(240,237,204,0.2)]" aria-label="Delete">
              <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                <path d="M3 6h18"></path>
                <path d="M19 6v14a2 2 0 01-2 2H7a2 2 0 01-2-2V6m3 0V4a2 2 0 012-2h4a2 2 0 012 2v2"></path>
              </svg>
            </button>
          </li>
        `;
          } else if (type === 'image') {
            // For images, show thumbnail
            html += `
          <li onclick="loadPreviousActivity('image', '${item.id}')" class="rounded hover:bg-[rgba(240,237,204,0.1)] p-2">
            <div class="flex items-center gap-2">
              <img src="${item.content}" class="w-12 h-12 object-cover rounded" alt="Thumbnail">
              <span class="text-xs text-[rgba(240,237,204,0.7)]">${formattedDate}</span>
            </div>
            <button onclick="event.stopPropagation(); deleteActivity('image', ${item.id})" class="p-1 rounded-full hover:bg-[rgba(240,237,204,0.2)]" aria-label="Delete">
              <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
                <path d="M3 6h18"></path>
                <path d="M19 6v14a2 2 0 01-2 2H7a2 2 0 01-2-2V6m3 0V4a2 2 0 012-2h4a2 2 0 012 2v2"></path>
              </svg>
            </button>
          </li>
        `;
          }
        });

        html += '</ul>';
        container.innerHTML = html;
      } catch (e) {
        console.error('Error updating recent sidebar:', e);
      }
    }

    function loadPreviousActivity(type, id) {
      try {
        const activities = JSON.parse(localStorage.getItem(`recent-${type}`));
        const activity = activities.find(a => a.id.toString() === id.toString());

        if (!activity) return;

        // Select the appropriate service
        const serviceOption = document.getElementById(`option-${type}`);
        selectService(type, serviceOption);

        if (type === 'text') {
          document.getElementById('grammar-input').value = activity.content;
          document.getElementById('grammar-form').querySelector('button[type="submit"]').disabled = false;
        } else if (type === 'image') {
          const preview = document.getElementById('image-preview');
          const previewContainer = document.getElementById('image-preview-container');
          preview.src = activity.content;
          previewContainer.classList.remove('hidden');
          document.getElementById('analyze-button').disabled = false;
        }

        // Close sidebar
        toggleRecentSidebar();
      } catch (e) {
        console.error('Error loading previous activity:', e);
      }
    }

    function deleteActivity(type, id) {
      try {
        const activities = JSON.parse(localStorage.getItem(`recent-${type}`));
        const updatedActivities = activities.filter(a => a.id !== id);
        localStorage.setItem(`recent-${type}`, JSON.stringify(updatedActivities));
        updateRecentSidebar(type);
      } catch (e) {
        console.error('Error deleting activity:', e);
      }
    }

      createParticles();
      loadRecentActivities();

      selectService('grammar', defaultOption);

        // Ctrl+Enter or Cmd+Enter to submit form

          if (currentService === 'grammar' && !grammarSubmitBtn.disabled) {
            grammarForm.requestSubmit();
          } else if (currentService === 'image' && !document.getElementById('analyze-button').disabled) {
            document.getElementById('image-form').requestSubmit();
          }
        }
      });

      // Add offline support notification
      window.addEventListener('online', () => showToast('You are back online!', 3000));
      window.addEventListener('offline', () => showToast('You are offline. Some features may be limited.', 5000));
    });
  </script>

          <label for="image-input" class="btn cursor-pointer inline-block mb-2">
            <i class="fas fa-upload"></i> Upload Image
          </label>
          <p class="text-sm opacity-70">or drag and drop your image here</p>
        </div>
        
        <div class="flex items-center justify-between mt-4">
          <button type="submit" id="analyze-image-btn" class="btn opacity-50 pointer-events-none" disabled>
            <i class="fas fa-check-circle"></i> Analyze Image
          </button>
          <button type="button" onclick="showRecent('image')" class="btn-outline btn">
            <i class="fas fa-history"></i> Recent Images
          </button>
        </div>
      </form>
      
      <!-- Loading Indicator -->
      <div id="image-loading" class="loading-indicator">
        <div class="loading-brain"></div>
        <div class="flex items-center">
          Processing your image
          <div class="thinking-dots ml-2">
            <span></span>

      <div class="mt-6">

          <button class="copy-btn" onclick="copyResults('image-results')">
            <i class="fas fa-copy"></i> Copy
          </button>
          <div id="image-results"></div>
        </div>
      </div>

  <!-- Recent Activity Modal -->
  <div id="recent-modal" class="modal p-6">
    <div class="flex justify-between items-center mb-6">
      <h2 id="modal-title" class="text-xl font-bold"></h2>
      <button onclick="closeModal()" class="text-[var(--accent-color)] text-lg" aria-label="Close modal">
        <i class="fas fa-times"></i>
      </button>
    </div>
    <div id="modal-content" class="space-y-3 max-h-[80vh] overflow-y-auto"></div>
    <button onclick="closeModal()" class="btn w-full mt-6">
      <i class="fas fa-times-circle"></i> Close
    </button>
  </div>

  <script>
    // Sidebar Toggle for Mobile
    document.getElementById('menu-toggle').addEventListener('click', () => {
      document.getElementById('sidebar').classList.toggle('active');
    });

    // Section Switching
    document.querySelectorAll('nav a[data-section]').forEach(link => {
      link.addEventListener('click', (e) => {
        e.preventDefault();
        document.querySelectorAll('.nav-link').forEach(navLink => navLink.classList.remove('active'));
        link.classList.add('active');
        switchSection(link.getAttribute('data-section'));
      });
    });

    function switchSection(sectionId) {
      document.querySelectorAll('.section').forEach(section => section.classList.remove('active'));
      document.getElementById(sectionId).classList.add('active');
      document.querySelectorAll('.nav-link').forEach(link => {
        if (link.getAttribute('data-section') === sectionId) {
          link.classList.add('active');
        } else {
          link.classList.remove('active');
        }
      });
      if (window.innerWidth < 768) document.getElementById('sidebar').classList.remove('active');
    }

    // Logout functionality
    document.getElementById('logout-btn').addEventListener('click', (e) => {
      e.preventDefault();
      showToast('Logging out...', 'success');
      setTimeout(() => window.location.href = '/logout', 1000);
    });

    // Toast Notification
    function showToast(message, type = 'info') {
      const toast = document.getElementById('toast');
      const toastMessage = document.getElementById('toast-message');
      toastMessage.textContent = message;
      toast.style.borderLeftColor = type === 'success' ? 'var(--success-color)' : type === 'error' ? 'var(--error-color)' : 'var(--accent-color)';
      toast.classList.add('active');
      setTimeout(() => toast.classList.remove('active'), 3000);
    }

    // Copy Results to Clipboard
    function copyResults(elementId) {
      const element = document.getElementById(elementId);
      if (element) {
        const text = element.innerText;
        navigator.clipboard.writeText(text).then(() => {
          showToast('Results copied to clipboard', 'success');
        }).catch(() => {
          showToast('Failed to copy results', 'error');
        });
      }
    }

    // Preview Image
    function previewImage(input) {
      const file = input.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
          document.getElementById('image-preview').innerHTML = `<img src="${e.target.result}" class="max-h-64 object-contain rounded-lg" />`;
          document.getElementById('analyze-image-btn').disabled = false;
          document.getElementById('analyze-image-btn').classList.remove('opacity-50', 'pointer-events-none');
        };
        reader.readAsDataURL(file);
      }
    }

    // HTMX Response Handling
    document.getElementById('text-form').addEventListener('htmx:response', function(event) {
      if (event.detail.successful) {
        document.getElementById('text-results-container').classList.remove('hidden');
        storeRecentText();
      }
    });

    document.getElementById('image-form').addEventListener('htmx:response', function(event) {
      if (event.detail.successful) {
        document.getElementById('image-results-container').classList.remove('hidden');
        storeRecentImage();
      }
    });

    // HTMX Error Handling
    document.body.addEventListener('htmx:responseError', (event) => {
      showToast('An error occurred during processing', 'error');
    });

    // Recent Activity Modal functions
    function showRecent(type) {
      const modal = document.getElementById('recent-modal');
      const title = document.getElementById('modal-title');
      const content = document.getElementById('modal-content');
      const items = getRecentItems(type);
      title.innerHTML = `<i class="fas fa-${type === 'text' ? 'font' : 'image'}" style="color: var(--accent-color)"></i> Recent ${type.charAt(0).toUpperCase() + type.slice(1)}s`;
      if (items.length === 0) {
        content.innerHTML = `<p class="opacity-70">No recent ${type}s found.</p>`;
      } else {
        if (type === 'text') {
          content.innerHTML = items.map((item, index) => `
            <div class="p-3 bg-[var(--panel-bg)] rounded-lg border border-[rgba(0,255,255,0.1)] cursor-pointer hover:border-[var(--accent-color)] transition-colors" onclick="reuseItem('${type}', ${index})">
              <p class="text-sm">${item.text.substring(0, 100)}${item.text.length > 100 ? '...' : ''}</p>
              <div class="flex justify-between items-center mt-2 text-xs opacity-70">
                <span>${item.date}</span>
                <button class="text-[var(--accent-color)] hover:underline" onclick="removeItem(event, '${type}', ${index})">Remove</button>
              </div>
            </div>
          `).join('');
        } else {
          content.innerHTML = `<div class="grid grid-cols-2 gap-3">` + 
            items.map((item, index) => `
              <div class="bg-[var(--panel-bg)] rounded-lg border border-[rgba(0,255,255,0.1)] overflow-hidden cursor-pointer hover:border-[var(--accent-color)] transition-colors" onclick="reuseItem('${type}', ${index})">
                <img src="${item.url}" class="w-full h-32 object-cover" alt="Recent image" />
                <div class="p-2 text-xs">
                  <div class="opacity-70">${item.date}</div>
                  <button class="text-[var(--accent-color)] hover:underline mt-1" onclick="removeItem(event, '${type}', ${index})">Remove</button>
                </div>
              </div>
            `).join('') + `</div>`;
        }
      }
      modal.classList.add('active');
    }

    function closeModal() {
      document.getElementById('recent-modal').classList.remove('active');
    }

    function removeItem(event, type, index) {
      event.stopPropagation();
      let items = JSON.parse(localStorage.getItem(`recent-${type}s`) || '[]');
      items.splice(index, 1);
      localStorage.setItem(`recent-${type}s`, JSON.stringify(items));
      showToast(`${type.charAt(0).toUpperCase() + type.slice(1)} removed from history`, 'success');
      showRecent(type);
      updateHomeRecentItems();
    }

    function reuseItem(type, index) {
      const items = getRecentItems(type);
      const item = items[index];
      if (type === 'text') {
        document.getElementById('text-input').value = item.text;
        switchSection('text-analysis');
      } else if (type === 'image') {
        document.getElementById('image-preview').innerHTML = `<img src="${item.url}" class="max-h-64 object-contain rounded-lg" />`;
        document.getElementById('analyze-image-btn').disabled = false;
        document.getElementById('analyze-image-btn').classList.remove('opacity-50', 'pointer-events-none');
        switchSection('image-analysis');
      }
      closeModal();
    }

    // Store Recent Text/Image in localStorage
    function storeRecentText() {
      const text = document.getElementById('text-input').value;
      if (!text.trim()) return;
      const newItem = {
        text: text,
        date: new Date().toLocaleString(),
        result: document.getElementById('text-results').innerHTML
      };
      let texts = JSON.parse(localStorage.getItem('recent-texts') || '[]');
      texts.unshift(newItem);
      if (texts.length > 10) texts.pop();
      localStorage.setItem('recent-texts', JSON.stringify(texts));
      updateHomeRecentItems();
    }

    function storeRecentImage() {
      const fileInput = document.getElementById('image-input');
      if (fileInput.files.length) {
        const file = fileInput.files[0];
        const reader = new FileReader();
        reader.onload = function(e) {
          const newItem = {
            url: e.target.result,
            date: new Date().toLocaleString(),
            result: document.getElementById('image-results').innerHTML
          };
          let images = JSON.parse(localStorage.getItem('recent-images') || '[]');
          images.unshift(newItem);
          if (images.length > 10) images.pop();
          localStorage.setItem('recent-images', JSON.stringify(images));
          updateHomeRecentItems();
        };
        reader.readAsDataURL(file);
      }
    }

    function getRecentItems(type) {
      return JSON.parse(localStorage.getItem(`recent-${type}s`) || '[]');
    }

    function updateHomeRecentItems() {
      const textsContainer = document.getElementById('home-recent-texts');
      const imagesContainer = document.getElementById('home-recent-images');
      const texts = getRecentItems('text');
      const images = getRecentItems('image');
      if (texts.length > 0) {
        textsContainer.innerHTML = texts.slice(0, 3).map((item, index) => `
          <div class="text-sm p-2 bg-[var(--panel-bg)] rounded border border-[rgba(0,255,255,0.1)] cursor-pointer hover:border-[var(--accent-color)]" onclick="reuseItem('text', ${index})">
            ${item.text.substring(0, 50)}${item.text.length > 50 ? '...' : ''}
          </div>
        `).join('');
      } else {
        textsContainer.innerHTML = `<p class="text-sm opacity-70">No recent text analyses</p>`;
      }
      if (images.length > 0) {
        imagesContainer.innerHTML = images.slice(0, 3).map((item, index) => `
          <div class="cursor-pointer" onclick="reuseItem('image', ${index})">
            <img src="${item.url}" class="w-full h-20 object-cover rounded" alt="Recent image" />
          </div>
        `).join('');
      } else {
        imagesContainer.innerHTML = `<p class="text-sm opacity-70 col-span-3">No recent image analyses</p>`;
      }
    }

    // Close modal when clicking outside
    document.getElementById('recent-modal').addEventListener('click', function(event) {
      if (event.target === this) {
        closeModal();
      }
    });

    // Initialize home recent items
    updateHomeRecentItems();
  </script>
</body>
